📊 5 fact tables · SQL DDL

-- Shrimp Aquaculture Industry Database · Schema (PostgreSQL 15+)

CREATE TABLE literature (
  id BIGSERIAL PRIMARY KEY,
  doi VARCHAR(120) UNIQUE,
  title TEXT NOT NULL,
  authors TEXT[],
  year SMALLINT,
  journal VARCHAR(200),
  source_db VARCHAR(40) NOT NULL, -- WoS / Scopus / CNKI / ...
  url TEXT,
  abstract TEXT,
  ingested_at TIMESTAMPTZ DEFAULT now(),
  merged_by VARCHAR(40),
  merged_at TIMESTAMPTZ,
  merger_version VARCHAR(20)
);

CREATE TABLE water_quality_fact (
  id BIGSERIAL PRIMARY KEY,
  lit_id BIGINT REFERENCES literature(id),
  mode VARCHAR(60),
  region VARCHAR(120),
  lat DECIMAL(9,6),
  lng DECIMAL(9,6),
  wt_c DECIMAL(5,2),
  salinity DECIMAL(5,2),
  do_mg_l DECIMAL(5,2),
  ph DECIMAL(4,2),
  nh3n_mg_l DECIMAL(8,3),
  no3n_mg_l DECIMAL(8,3),
  no2n_mg_l DECIMAL(8,3),
  cod_mg_l DECIMAL(8,2),
  tn_mg_l DECIMAL(8,2),
  tp_mg_l DECIMAL(8,2),
  tss_mg_l DECIMAL(8,2),
  sample_date DATE,
  evidence_blob TEXT,
  confidence DECIMAL(3,2),
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE growth_fact (
  id BIGSERIAL PRIMARY KEY,
  lit_id BIGINT REFERENCES literature(id),
  mode VARCHAR(60),
  region VARCHAR(120),
  initial_w_g DECIMAL(8,3),
  final_w_g DECIMAL(8,3),
  survival_pct DECIMAL(5,2),
  sgr_pct_day DECIMAL(5,2),
  fcr DECIMAL(4,2),
  yield_kg_m3 DECIMAL(8,2),
  evidence_blob TEXT,
  confidence DECIMAL(3,2)
);

CREATE TABLE cost_fact (
  id BIGSERIAL PRIMARY KEY,
  lit_id BIGINT REFERENCES literature(id),
  mode VARCHAR(60),
  item VARCHAR(60), -- seed / feed / health products / labour / utilities / ...
  cost_yuan_per_mu DECIMAL(12,2),
  pct_of_total DECIMAL(5,2),
  year SMALLINT
);

CREATE TABLE benefit_fact (
  id BIGSERIAL PRIMARY KEY,
  lit_id BIGINT REFERENCES literature(id),
  mode VARCHAR(60),
  total_cost DECIMAL(12,2),
  total_revenue DECIMAL(12,2),
  net_profit DECIMAL(12,2),
  profit_rate DECIMAL(5,2),
  breakeven_price DECIMAL(8,2),
  yield_kg_per_mu DECIMAL(10,2),
  year SMALLINT
);

CREATE TABLE env_fact (
  id BIGSERIAL PRIMARY KEY,
  lit_id BIGINT REFERENCES literature(id),
  impact_type VARCHAR(60), -- global warming / eutrophication / acidification
  unit_factor VARCHAR(30), -- kg CO2 eq / kg PO4³⁻ / ...
  substance VARCHAR(40),
  factor_value DECIMAL(8,3),
  emissions_kg DECIMAL(10,3)
);

CREATE INDEX idx_wq_lit ON water_quality_fact(lit_id);
CREATE INDEX idx_wq_mode ON water_quality_fact(mode);
CREATE INDEX idx_growth_lit ON growth_fact(lit_id);
CREATE INDEX idx_cost_mode ON cost_fact(mode,year);
CREATE INDEX idx_lit_year ON literature(year);

⚙️ Unit conversion and data-quality conventions

Water quality

  • mg/L = ppm (approximately 1:1 in fresh water / low salinity)
  • mg/L × molecular weight / 1000 = mmol/L
  • NH₃-N + NH₄⁺-N = TAN (Total Ammonia Nitrogen)
  • Temperature and pH jointly determine NH₃ toxicity

Cost / yield

  • 1 mu = 666.67 m²
  • 1 CNY/kg = 1000 CNY/t
  • 1 kg/m³ = 0.0624 lb/gal (US)
  • FCR = feed intake / weight gain
